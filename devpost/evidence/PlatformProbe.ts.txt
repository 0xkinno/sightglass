import { Gemini } from "Remote Service Gateway.lspkg/HostedExternal/Gemini";
import { DIAL_PNG_B64 } from "./DialB64";

@component
export class PlatformProbe extends BaseScriptComponent {
  @input cropTexture: Texture;

  private internetModule: InternetModule = require("LensStudio:InternetModule");
  private cameraModule: CameraModule = require("LensStudio:CameraModule");

  onAwake() {
    this.createEvent("OnStartEvent").bind(() => {
      this.probeGemini();
      this.probePost();
      this.probeCamera();
      this.probeVisionDial();
    });
  }

  private geminiJson(parts: any[]): Promise<any> {
    return Gemini.models({
      model: "gemini-2.5-flash",
      type: "generateContent",
      body: {
        contents: [{ role: "user", parts }],
        generationConfig: { temperature: 0, responseMimeType: "application/json" },
      },
    } as any);
  }

  private probeGemini() {
    const t0 = Date.now();
    this.geminiJson([{ text: 'Return JSON {"word":"ready"}' }])
      .then((r: any) => print("PROBE_A ok model=gemini-2.5-flash ms=" + (Date.now() - t0) + " text=" + JSON.stringify(r?.candidates?.[0]?.content?.parts?.[0]?.text)))
      .catch((e: any) => print("PROBE_A fail ms=" + (Date.now() - t0) + " err=" + String(e).slice(0, 300)));
  }

  private async probePost() {
    const t0 = Date.now();
    try {
      const res = await this.internetModule.fetch("https://httpbin.org/post", {
        method: "POST",
        body: JSON.stringify({ probe: "sightglass" }),
        headers: { "Content-Type": "application/json" },
      });
      print("PROBE_D status=" + res.status + " ms=" + (Date.now() - t0));
    } catch (e) {
      print("PROBE_D fail ms=" + (Date.now() - t0) + " err=" + String(e));
    }
  }

  private async probeCamera() {
    try {
      const req = CameraModule.createCameraRequest();
      req.cameraId = CameraModule.CameraId.Default_Color;
      const camTex = this.cameraModule.requestCamera(req);
      const provider = camTex.control as CameraTextureProvider;
      await new Promise<void>((resolve) => {
        const reg = provider.onNewFrame.add(() => {
          provider.onNewFrame.remove(reg);
          resolve();
        });
      });
      const crop = this.cropTexture.control as any;
      crop.inputTexture = camTex;
      crop.cropRect.left = -0.4;
      crop.cropRect.right = 0.4;
      crop.cropRect.bottom = -0.4;
      crop.cropRect.top = 0.4;
      await new Promise<void>((resolve) => {
        const reg = provider.onNewFrame.add(() => {
          provider.onNewFrame.remove(reg);
          resolve();
        });
      });
      const t0 = Date.now();
      const b64: string = await new Promise((resolve, reject) => {
        Base64.encodeTextureAsync(this.cropTexture, (s) => resolve(s), () => reject("encode failed"), CompressionQuality.HighQuality, EncodingType.Jpg);
      });
      const bytes = Base64.decode(b64) as unknown as Uint8Array;
      const seen = new Set<number>();
      for (let i = 0; i < bytes.length; i += 7) seen.add(bytes[i]);
      print("PROBE_B bytes=" + bytes.length + " distinctByteValues=" + seen.size + " ms=" + (Date.now() - t0) + " nonblank=" + (bytes.length > 4000 && seen.size > 150));
      // (c) vision on the composite crop
      const t1 = Date.now();
      const r = await this.geminiJson([
        { inlineData: { mimeType: "image/jpeg", data: b64 } },
        { text: 'Describe what is in this image. Return JSON {"subject": string, "person_visible": boolean}' },
      ]);
      const txt = r?.candidates?.[0]?.content?.parts?.[0]?.text;
      const parsed = JSON.parse(txt);
      print("PROBE_C_CROP ok ms=" + (Date.now() - t1) + " parsed=" + JSON.stringify(parsed));
    } catch (e) {
      print("PROBE_B_or_C_CROP fail err=" + String(e).slice(0, 300));
    }
  }

  private async probeVisionDial() {
    const t0 = Date.now();
    try {
      const r = await this.geminiJson([
        { inlineData: { mimeType: "image/png", data: DIAL_PNG_B64 } },
        { text: 'This is a pressure dial. Read the needle. Return JSON {"reading": number, "unit": string}' },
      ]);
      const txt = r?.candidates?.[0]?.content?.parts?.[0]?.text;
      const parsed = JSON.parse(txt);
      print("PROBE_C_DIAL ok ms=" + (Date.now() - t0) + " parsed=" + JSON.stringify(parsed) + " expected=70 bar");
    } catch (e) {
      print("PROBE_C_DIAL fail ms=" + (Date.now() - t0) + " err=" + String(e).slice(0, 300));
    }
  }
}
